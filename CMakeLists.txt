cmake_minimum_required(VERSION 3.24)
project(CarDiagnosticAnalysis VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(CDA_BUILD_ENGINE "Build native platform-engine acceleration library" ON)
option(CDA_BUILD_UI "Build shared Qt Quick client" ON)
option(CDA_BUILD_BACKEND "Build local macroservice host" ON)
option(CDA_SANITIZE "Enable address/undefined sanitizers" OFF)
option(CDA_COVERAGE "Enable GCC coverage" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options(-Wall -Wextra -Wpedantic)
 if(CDA_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
 endif()
 if(CDA_COVERAGE)
  add_compile_options(--coverage -O0 -g)
  add_link_options(--coverage)
 endif()
endif()
find_package(OpenSSL 3 REQUIRED)
add_library(cda_core core/protocols/elm.cpp core/security/write_policy.cpp core/security/crypto.cpp)
target_include_directories(cda_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(cda_core PUBLIC OpenSSL::Crypto)
include(CTest)
if(BUILD_TESTING)
 add_executable(core_tests tests/unit/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE cda_core)
 add_test(NAME core_contracts COMMAND core_tests)
 add_executable(protocol_benchmark tests/performance/protocol_benchmark.cpp)
 target_link_libraries(protocol_benchmark PRIVATE cda_core)
endif()
if(CDA_BUILD_UI OR CDA_BUILD_BACKEND)
 set(CMAKE_AUTOMOC ON)
 find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Sql Test Concurrent)
 add_library(cda_services services/identity/store.cpp core/transport/qt_elm.cpp services/diagnostics/controller.cpp)
 target_include_directories(cda_services PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(cda_services PUBLIC cda_core Qt6::Core Qt6::Network Qt6::Sql Qt6::Concurrent)
 if(NOT ANDROID AND NOT IOS)
  find_package(Qt6 REQUIRED COMPONENTS SerialPort)
  target_link_libraries(cda_services PUBLIC Qt6::SerialPort)
 endif()
 if(BUILD_TESTING)
  add_executable(service_tests tests/integration/service_tests.cpp)
  target_link_libraries(service_tests PRIVATE cda_services Qt6::Test)
  add_test(NAME service_contracts COMMAND service_tests)
 endif()
endif()
if(CDA_BUILD_UI)
 find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Svg LinguistTools)
 qt_standard_project_setup(REQUIRES 6.8)
 qt_add_executable(car_diagnostic apps/shared/main.cpp apps/shared/app_model.cpp apps/shared/platform_client.cpp)
 if(WIN32)
  configure_file(apps/windows/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
  target_sources(car_diagnostic PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
 endif()
 if(ANDROID)
  set_target_properties(car_diagnostic PROPERTIES QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/apps/android")
 endif()
 target_link_libraries(car_diagnostic PRIVATE cda_services Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
 if(BUILD_TESTING)
  target_compile_definitions(car_diagnostic PRIVATE CDA_UI_CHECKS)
  target_link_libraries(car_diagnostic PRIVATE Qt6::Test)
 endif()
 set_source_files_properties(ui/themes/Neo.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
 file(GLOB NAV_ICONS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/ui/icons/*.svg")
 qt_add_qml_module(car_diagnostic URI CarDiagnostic VERSION 1.0
  QML_FILES ui/pages/PlatformPage.qml ui/Main.qml ui/components/MetricCard.qml ui/themes/Neo.qml ui/charts/TelemetryTrace.qml
   ui/pages/LauncherPage.qml ui/components/NeoButton.qml ui/components/NeoComboBox.qml
   ui/components/NeoTextField.qml ui/components/NeoTextArea.qml ui/components/NeoSpinBox.qml ui/components/NeoDialog.qml
  RESOURCES assets/platform-operations.json assets/app_icon/canonical.svg assets/launcher/background.png assets/launcher/icons-3d.png assets/launcher/brand-car.png ${NAV_ICONS})
 set_target_properties(car_diagnostic PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE
  MACOSX_BUNDLE_BUNDLE_NAME "Car Diagnostic Analysis"
  MACOSX_BUNDLE_GUI_IDENTIFIER "org.cardiagnostic.analysis")
 qt_add_translations(car_diagnostic TS_FILES ui/translations/cda_hy.ts ui/translations/cda_ru.ts ui/translations/cda_fr.ts ui/translations/cda_en.ts RESOURCE_PREFIX "/i18n")
 install(TARGETS car_diagnostic BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
if(CDA_BUILD_BACKEND)
 find_package(Qt6 REQUIRED COMPONENTS HttpServer)
 qt_add_executable(cda_backend backend/api/main.cpp)
 target_link_libraries(cda_backend PRIVATE cda_services Qt6::HttpServer)
endif()

if(CDA_BUILD_ENGINE)
 find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
 add_library(cda_engine SHARED native/engine/bridge.cpp)
 target_link_libraries(cda_engine PRIVATE cda_core Qt6::Core)
 install(TARGETS cda_engine LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
endif()
