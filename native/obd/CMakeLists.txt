cmake_minimum_required(VERSION 3.20)
project(ai_car_obd LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
if(WIN32)
  message(FATAL_ERROR "Native TCP transport currently targets POSIX. Use the existing Python transport on Windows.")
endif()
pybind11_add_module(_obd_native tcp.cpp)
target_compile_features(_obd_native PRIVATE cxx_std_20)
target_compile_options(_obd_native PRIVATE -Wall -Wextra -Wpedantic)
